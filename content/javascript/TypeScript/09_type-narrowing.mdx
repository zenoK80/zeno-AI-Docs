## typeof 타입 가드

```ts
function processValue(value: string | number | boolean): string {
  if (typeof value === 'string') {
    return value.toUpperCase()       // string 메서드 사용 가능
  } else if (typeof value === 'number') {
    return value.toFixed(2)          // number 메서드 사용 가능
  } else {
    return value ? '참' : '거짓'     // boolean
  }
}

// typeof로 구분 가능한 타입
// 'string' | 'number' | 'bigint' | 'boolean' | 'symbol' | 'undefined' | 'object' | 'function'
```

## instanceof 타입 가드

```ts
class Dog {
  bark() { return '멍멍' }
}

class Cat {
  meow() { return '야옹' }
}

function makeSound(animal: Dog | Cat): string {
  if (animal instanceof Dog) {
    return animal.bark()   // Dog 타입
  } else {
    return animal.meow()   // Cat 타입
  }
}
```

## in 연산자 타입 가드

```ts
interface Fish {
  swim(): void
}

interface Bird {
  fly(): void
}

function move(creature: Fish | Bird): void {
  if ('swim' in creature) {
    creature.swim()   // Fish
  } else {
    creature.fly()    // Bird
  }
}

// 판별 유니온과 결합
type NetworkResponse =
  | { status: 'success'; data: unknown }
  | { status: 'error'; message: string }

function handleResponse(response: NetworkResponse) {
  if ('data' in response) {
    console.log(response.data)     // success 타입
  } else {
    console.error(response.message) // error 타입
  }
}
```

## 사용자 정의 타입 가드 함수

반환 타입에 `is` 키워드를 사용합니다.

```ts
// value is string — 타입 서술어 (type predicate)
function isString(value: unknown): value is string {
  return typeof value === 'string'
}

function isUser(obj: unknown): obj is User {
  return (
    typeof obj === 'object' &&
    obj !== null &&
    'id' in obj &&
    'name' in obj
  )
}

interface User {
  id: number
  name: string
}

// 사용
function processUser(data: unknown) {
  if (isUser(data)) {
    console.log(data.name)  // TypeScript가 User 타입으로 인식
  }
}
```

## 단언 함수 (Assertion Function)

조건이 거짓이면 예외를 던지고 타입을 좁힙니다.

```ts
function assert(condition: unknown, message: string): asserts condition {
  if (!condition) {
    throw new Error(message)
  }
}

function assertIsString(value: unknown): asserts value is string {
  if (typeof value !== 'string') {
    throw new Error('문자열이 아닙니다.')
  }
}

// 사용
function processInput(input: string | null) {
  assert(input !== null, '입력값이 null입니다.')
  // 여기서부터 input은 string 타입
  console.log(input.toUpperCase())
}
```

## 완전성 검사 (Exhaustiveness Check)

모든 경우를 처리했는지 컴파일 타임에 확인합니다.

```ts
type Shape = 'circle' | 'square' | 'triangle'

function describeShape(shape: Shape): string {
  switch (shape) {
    case 'circle': return '원'
    case 'square': return '사각형'
    case 'triangle': return '삼각형'
    default:
      const _exhaustive: never = shape
      throw new Error(`알 수 없는 도형: ${_exhaustive}`)
  }
}
// 새 Shape 타입 추가 시 default에서 컴파일 오류 발생 → 처리 누락 방지
```

## 타입 축소 패턴 정리

```ts
// null/undefined 처리
function process(value: string | null | undefined) {
  if (value == null) return  // null과 undefined 모두 처리
  value.toUpperCase()        // 여기서는 string
}

// 진실성 검사
function printLength(value: string | null | undefined) {
  if (value) {
    console.log(value.length)  // truthy이면 string
  }
}

// 동등 비교
type Status = 'a' | 'b' | 'c'

function handle(status: Status) {
  if (status === 'a') {
    // 'a'만
  } else {
    // 'b' | 'c'
  }
}
```

## 핵심 정리

- `typeof`로 원시 타입을, `instanceof`로 클래스 인스턴스를 구별합니다.
- `in` 연산자로 프로퍼티 존재 여부로 타입을 좁힙니다.
- 사용자 정의 타입 가드 `value is Type`으로 복잡한 타입 검사를 캡슐화합니다.
- never 타입을 활용한 완전성 검사로 누락 케이스를 컴파일 타임에 발견합니다.
