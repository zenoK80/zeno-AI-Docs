## Express 타입 활용

```ts
import express, { Request, Response, NextFunction } from 'express'

const app = express()
app.use(express.json())

// 기본 라우트
app.get('/health', (req: Request, res: Response) => {
  res.json({ status: 'ok' })
})

// 경로 매개변수 타입
app.get('/users/:id', (req: Request<{ id: string }>, res: Response) => {
  const { id } = req.params
  res.json({ userId: id })
})

// 쿼리 파라미터 타입
interface UserQuery {
  page?: string
  limit?: string
  search?: string
}

app.get('/users', (req: Request<{}, {}, {}, UserQuery>, res: Response) => {
  const { page = '1', limit = '10', search } = req.query
  res.json({ page: Number(page), limit: Number(limit), search })
})

// 요청 본문 타입
interface CreateUserBody {
  name: string
  email: string
  role?: 'admin' | 'user'
}

app.post('/users', (req: Request<{}, {}, CreateUserBody>, res: Response) => {
  const { name, email, role = 'user' } = req.body
  res.status(201).json({ id: Date.now(), name, email, role })
})
```

## 환경 변수 타이핑

```ts
// types/env.d.ts
declare namespace NodeJS {
  interface ProcessEnv {
    NODE_ENV: 'development' | 'production' | 'test'
    PORT: string
    DATABASE_URL: string
    JWT_SECRET: string
    REDIS_URL?: string
  }
}

// env.ts — 환경 변수 검증 및 타입 변환
interface AppConfig {
  port: number
  databaseUrl: string
  jwtSecret: string
  isProduction: boolean
}

function loadConfig(): AppConfig {
  const { PORT, DATABASE_URL, JWT_SECRET, NODE_ENV } = process.env

  if (!DATABASE_URL) throw new Error('DATABASE_URL 환경 변수가 필요합니다.')
  if (!JWT_SECRET) throw new Error('JWT_SECRET 환경 변수가 필요합니다.')

  return {
    port: Number(PORT) || 3000,
    databaseUrl: DATABASE_URL,
    jwtSecret: JWT_SECRET,
    isProduction: NODE_ENV === 'production',
  }
}

export const config = loadConfig()
```

## 에러 핸들링 미들웨어

```ts
interface ApiError {
  status: number
  message: string
  errors?: Record<string, string>
}

function errorHandler(
  error: Error,
  req: Request,
  res: Response,
  next: NextFunction
): void {
  console.error(error.stack)

  if (error instanceof ValidationError) {
    res.status(400).json({
      status: 400,
      message: error.message,
      errors: error.fields,
    } satisfies ApiError)
    return
  }

  res.status(500).json({
    status: 500,
    message: '서버 내부 오류가 발생했습니다.',
  } satisfies ApiError)
}

class ValidationError extends Error {
  constructor(message: string, public fields: Record<string, string>) {
    super(message)
    this.name = 'ValidationError'
  }
}
```

## path alias 설정 (tsconfig-paths)

```bash
npm install -D tsconfig-paths
```

```json
// package.json
{
  "scripts": {
    "dev": "nodemon --exec ts-node -r tsconfig-paths/register src/index.ts",
    "build": "tsc",
    "start": "node -r ./dist/register.js dist/index.js"
  }
}
```

```ts
// src/register.ts — 프로덕션에서 경로 별칭 처리
import { register } from 'tsconfig-paths'
import { compilerOptions } from '../tsconfig.json'

register({
  baseUrl: compilerOptions.baseUrl || '.',
  paths: compilerOptions.paths || {},
})
```

## 핵심 정리

- `@types/node`와 `@types/express`로 Node.js와 Express 타입 지원을 추가합니다.
- `Request<Params, ResBody, ReqBody, Query>`로 Express 요청의 각 부분 타입을 지정합니다.
- `declare namespace NodeJS` 확장으로 process.env에 타입을 제공합니다.
- 경로 별칭은 tsconfig.json의 paths와 tsconfig-paths로 런타임에서도 동작하게 합니다.
