## Partial - 모든 프로퍼티를 선택적으로

```ts
interface User {
  id: number
  name: string
  email: string
  age: number
}

type PartialUser = Partial<User>
// { id?: number; name?: string; email?: string; age?: number }

// 업데이트 함수에서 유용
function updateUser(id: number, updates: Partial<User>) {
  // 일부 필드만 업데이트 가능
}

updateUser(1, { name: 'Bob' })         // OK
updateUser(1, { email: 'b@test.com' }) // OK
```

## Required - 모든 프로퍼티를 필수로

```ts
interface Config {
  host?: string
  port?: number
  debug?: boolean
}

type RequiredConfig = Required<Config>
// { host: string; port: number; debug: boolean }

// Partial의 반대
```

## Readonly - 모든 프로퍼티를 읽기 전용으로

```ts
interface Config {
  host: string
  port: number
}

type ReadonlyConfig = Readonly<Config>

const config: ReadonlyConfig = { host: 'localhost', port: 3000 }
config.host = 'other'  // 오류: 읽기 전용
```

## Pick - 특정 프로퍼티만 선택

```ts
interface User {
  id: number
  name: string
  email: string
  password: string
  createdAt: Date
}

// API 응답에서 민감 정보 제외
type PublicUser = Pick<User, 'id' | 'name' | 'email'>
// { id: number; name: string; email: string }

function getPublicProfile(user: User): PublicUser {
  return {
    id: user.id,
    name: user.name,
    email: user.email,
  }
}
```

## Omit - 특정 프로퍼티를 제외

```ts
// 생성 시 id와 createdAt은 서버에서 생성
type CreateUserDto = Omit<User, 'id' | 'createdAt'>
// { name: string; email: string; password: string }

function createUser(dto: CreateUserDto): User {
  return {
    ...dto,
    id: generateId(),
    createdAt: new Date(),
  }
}
```

## Record - 키-값 쌍의 객체 타입

```ts
// Record<Keys, Values>
type UserMap = Record<string, User>
// { [key: string]: User }

const users: UserMap = {
  alice: { id: 1, name: 'Alice', email: 'a@test.com', password: 'pw', createdAt: new Date() },
}

// 유니온 키와 함께
type Status = 'pending' | 'active' | 'closed'
type StatusLabel = Record<Status, string>

const labels: StatusLabel = {
  pending: '대기 중',
  active: '활성',
  closed: '종료',
}
```

## ReturnType - 함수 반환 타입 추출

```ts
function fetchUser(id: number) {
  return { id, name: 'Alice', email: 'a@test.com' }
}

type FetchUserResult = ReturnType<typeof fetchUser>
// { id: number; name: string; email: string }

// async 함수는 Promise로 감싸짐
async function fetchData() {
  return { count: 10, items: [] }
}

type FetchDataResult = Awaited<ReturnType<typeof fetchData>>
// { count: number; items: never[] }
```

## Parameters - 함수 매개변수 타입 추출

```ts
function createPost(title: string, content: string, authorId: number) {
  return { title, content, authorId }
}

type CreatePostParams = Parameters<typeof createPost>
// [string, string, number]

// 특정 매개변수 타입 접근
type FirstParam = Parameters<typeof createPost>[0]  // string

// 사용 예시
function logCall<T extends (...args: unknown[]) => unknown>(
  fn: T,
  ...args: Parameters<T>
): ReturnType<T> {
  console.log('호출:', fn.name, args)
  return fn(...args) as ReturnType<T>
}
```

## NonNullable - null/undefined 제거

```ts
type MaybeString = string | null | undefined
type DefinitelyString = NonNullable<MaybeString>  // string

// 배열에서 null 제거
const items = ['a', null, 'b', undefined, 'c']
const filtered = items.filter((x): x is NonNullable<typeof x> => x != null)
// 타입: string[]
```

## 핵심 정리

- `Partial<T>`: 모든 프로퍼티를 선택적으로
- `Required<T>`: 모든 프로퍼티를 필수로
- `Readonly<T>`: 모든 프로퍼티를 읽기 전용으로
- `Pick<T, K>`: T에서 K 프로퍼티만 선택
- `Omit<T, K>`: T에서 K 프로퍼티를 제외
- `Record<K, V>`: 키-값 매핑 타입 생성
- `ReturnType<F>`: 함수 반환 타입 추출
- `Parameters<F>`: 함수 매개변수 타입 추출
