## Props 타입 정의

```tsx
// interface로 Props 정의
interface ButtonProps {
  label: string
  onClick: () => void
  disabled?: boolean
  variant?: 'primary' | 'secondary' | 'danger'
  children?: React.ReactNode
}

function Button({ label, onClick, disabled = false, variant = 'primary' }: ButtonProps) {
  return (
    <button
      onClick={onClick}
      disabled={disabled}
      className={`btn btn-${variant}`}
    >
      {label}
    </button>
  )
}
```

## useState 제네릭

```tsx
import { useState } from 'react'

// 타입 추론
const [count, setCount] = useState(0)           // number
const [name, setName] = useState('')             // string
const [active, setActive] = useState(false)      // boolean

// 명시적 타입 제공
const [user, setUser] = useState<User | null>(null)
const [items, setItems] = useState<string[]>([])
const [status, setStatus] = useState<'idle' | 'loading' | 'error'>('idle')

interface User {
  id: number
  name: string
}
```

      // 처리
    }
  }

## useRef 타입

```tsx
import { useRef, useEffect } from 'react'

function InputFocus() {
  // DOM 참조 — 초기값 null, 제네릭으로 DOM 타입 지정
  const inputRef = useRef<HTMLInputElement>(null)

  useEffect(() => {
    inputRef.current?.focus()  // null 체크 필요
  }, [])

  return <input ref={inputRef} />
}

// 변경 가능한 값 저장
const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
```

type CounterAction =
  | { type: 'increment' }
  | { type: 'decrement' }
  | { type: 'reset'; payload: number }

interface CounterState {
  count: number
}

function reducer(state: CounterState, action: CounterAction): CounterState {
  switch (action.type) {
    case 'increment': return { count: state.count + 1 }
    case 'decrement': return { count: state.count - 1 }
    case 'reset': return { count: action.payload }
    default: return state
  }
}

export default function Counter() {
  const [state, dispatch] = useReducer(reducer, { count: 0 })
  return (
    <div>
      <p>{state.count}</p>
      <button onClick={() => dispatch({ type: 'increment' })}>+</button>
      <button onClick={() => dispatch({ type: 'reset', payload: 0 })}>리셋</button>
    </div>
  )
}`,
  }}

## FC vs 함수 컴포넌트 직접 정의

```tsx
// FC (React.FC) 방식 — 현재는 잘 사용하지 않음
const Button: React.FC<ButtonProps> = ({ label, onClick }) => {
  return <button onClick={onClick}>{label}</button>
}
// 단점: children이 암묵적으로 포함됨 (React 18 이후 해결됨)

// 권장 방식 — 함수 컴포넌트 직접 타입 지정
function Button({ label, onClick }: ButtonProps): React.ReactElement {
  return <button onClick={onClick}>{label}</button>
}

// 또는 반환 타입 생략 (추론)
function Button({ label, onClick }: ButtonProps) {
  return <button onClick={onClick}>{label}</button>
}
```

## Context 타입

```tsx
interface ThemeContextType {
  theme: 'light' | 'dark'
  toggleTheme: () => void
}

const ThemeContext = createContext<ThemeContextType | null>(null)

function useTheme(): ThemeContextType {
  const context = useContext(ThemeContext)
  if (!context) throw new Error('ThemeProvider 안에서 사용해야 합니다.')
  return context
}
```

## 핵심 정리

- Props는 interface나 type alias로 정의하고 구조 분해 시 타입 적용합니다.
- `useState<T>`로 제네릭을 명시하면 초기값만으로 추론이 어려운 타입을 처리합니다.
- 이벤트 핸들러는 `React.ChangeEvent<HTMLInputElement>` 같은 React 이벤트 타입을 사용합니다.
- `useRef<HTMLElement>(null)`로 DOM 참조에 타입을 지정하고 `.current?.`으로 null 체크합니다.
