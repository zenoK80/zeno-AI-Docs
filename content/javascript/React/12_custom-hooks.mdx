## 커스텀 훅이란?

커스텀 훅은 `use`로 시작하는 일반 함수로, 내부에서 React 훅을 호출할 수 있습니다. 여러 컴포넌트에서 반복되는 상태 로직을 추출해 한 곳에서 관리합니다.

```js
// 커스텀 훅 기본 형태
function useCounter(initialValue = 0) {
  const [count, setCount] = useState(initialValue)

  const increment = () => setCount(c => c + 1)
  const decrement = () => setCount(c => c - 1)
  const reset = () => setCount(initialValue)

  return { count, increment, decrement, reset }
}

// 사용
function Counter() {
  const { count, increment, decrement, reset } = useCounter(10)
  return (
    <div>
      <p>{count}</p>
      <button onClick={increment}>+</button>
      <button onClick={decrement}>-</button>
      <button onClick={reset}>초기화</button>
    </div>
  )
}
```

## useFetch — 데이터 페칭 훅

```js
function useFetch(url) {
  const [data, setData] = useState(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)

  useEffect(() => {
    let cancelled = false

    setLoading(true)
    setError(null)

    fetch(url)
      .then(res => {
        if (!res.ok) throw new Error(`HTTP ${res.status}`)
        return res.json()
      })
      .then(json => {
        if (!cancelled) {
          setData(json)
          setLoading(false)
        }
      })
      .catch(err => {
        if (!cancelled) {
          setError(err.message)
          setLoading(false)
        }
      })

    return () => {
      cancelled = true
    }
  }, [url])

  return { data, loading, error }
}

// 사용
function UserProfile({ userId }) {
  const { data, loading, error } = useFetch(`/api/users/${userId}`)

  if (loading) return <p>로딩 중...</p>
  if (error) return <p>에러: {error}</p>
  return <p>{data.name}</p>
}
```

## useLocalStorage — 로컬스토리지 동기화 훅

```js
function useLocalStorage(key, initialValue) {
  const [storedValue, setStoredValue] = useState(() => {
    try {
      const item = window.localStorage.getItem(key)
      return item ? JSON.parse(item) : initialValue
    } catch {
      return initialValue
    }
  })

  const setValue = value => {
    try {
      const valueToStore = value instanceof Function ? value(storedValue) : value
      setStoredValue(valueToStore)
      window.localStorage.setItem(key, JSON.stringify(valueToStore))
    } catch (error) {
      console.error(error)
    }
  }

  return [storedValue, setValue]
}

// 사용
function ThemeToggle() {
  const [theme, setTheme] = useLocalStorage('theme', 'light')
  return (
    <button onClick={() => setTheme(t => (t === 'light' ? 'dark' : 'light'))}>
      현재 테마: {theme}
    </button>
  )
}
```

## useDebounce — 디바운스 훅

```js
function useDebounce(value, delay = 300) {
  const [debouncedValue, setDebouncedValue] = useState(value)

  useEffect(() => {
    const timer = setTimeout(() => {
      setDebouncedValue(value)
    }, delay)

    return () => clearTimeout(timer)
  }, [value, delay])

  return debouncedValue
}

// 사용 — 검색어 입력 최적화
function SearchBox() {
  const [query, setQuery] = useState('')
  const debouncedQuery = useDebounce(query, 500)

  useEffect(() => {
    if (debouncedQuery) {
      console.log('API 호출:', debouncedQuery)
    }
  }, [debouncedQuery])

  return (
    <input
      value={query}
      onChange={e => setQuery(e.target.value)}
      placeholder="검색어 입력..."
    />
  )
}
```

## useToggle — 불리언 토글 훅

```js
function useToggle(initialValue = false) {
  const [value, setValue] = useState(initialValue)
  const toggle = () => setValue(v => !v)
  const setTrue = () => setValue(true)
  const setFalse = () => setValue(false)
  return [value, toggle, setTrue, setFalse]
}

function Modal() {
  const [isOpen, toggleModal, openModal, closeModal] = useToggle()
  return (
    <div>
      <button onClick={openModal}>열기</button>
      {isOpen && (
        <div>
          <p>모달 내용</p>
          <button onClick={closeModal}>닫기</button>
        </div>
      )}
    </div>
  )
}
```

## 커스텀 훅 규칙

```
1. 이름은 반드시 use로 시작해야 합니다.
2. 최상위 레벨에서만 훅을 호출합니다 (조건문·반복문 내부 금지).
3. 커스텀 훅은 상태와 로직을 공유하지만, 상태 자체는 각 컴포넌트에 독립적으로 존재합니다.
4. 반환값의 형태는 자유롭게 설계합니다 (배열, 객체, 단일 값 모두 가능).
```

## 핵심 정리

- 커스텀 훅은 `use`로 시작하는 함수로, 내부에서 React 훅을 사용합니다.
- 반복되는 상태 로직을 컴포넌트에서 분리해 재사용성을 높입니다.
- `useFetch`, `useLocalStorage`, `useDebounce` 등이 대표적인 실전 패턴입니다.
- 상태 자체는 각 컴포넌트 인스턴스에 독립적으로 유지됩니다.
