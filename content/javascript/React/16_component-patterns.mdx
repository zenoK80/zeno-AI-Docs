## 컴포넌트 합성 (Composition)

상속 대신 **합성**으로 컴포넌트를 설계합니다. `children`이나 여러 slots(prop으로 전달하는 JSX)을 활용합니다.

### 기본 children 합성

```jsx
function Card({ title, footer, children }) {
  return (
    <div className="card">
      <div className="card-header">
        <h2>{title}</h2>
      </div>
      <div className="card-body">
        {children}
      </div>
      {footer && (
        <div className="card-footer">
          {footer}
        </div>
      )}
    </div>
  )
}

// 사용
function App() {
  return (
    <Card
      title="주문 정보"
      footer={<button>결제하기</button>}
    >
      <p>상품: 노트북</p>
      <p>수량: 1개</p>
      <p>가격: 1,200,000원</p>
    </Card>
  )
}
```

## Compound Component (복합 컴포넌트)

탭, 아코디언 같이 서로 연관된 여러 컴포넌트를 하나의 그룹으로 만드는 패턴입니다.

<CodePlayground
  template="react"
  activeFile="/App.js"
  label="Compound Component 실습 — 탭을 클릭해보세요"

    '/App.js': `import { createContext, useContext, useState } from 'react'

const TabContext = createContext(null)

function Tabs({ defaultTab, children }) {
  const [activeTab, setActiveTab] = useState(defaultTab)
  return (
    <TabContext.Provider value={{ activeTab, setActiveTab }}>
      <div className="tabs">{children}</div>
    </TabContext.Provider>
  )
}

function TabList({ children }) {
  return <div className="tab-list">{children}</div>
}

function Tab({ id, children }) {
  const { activeTab, setActiveTab } = useContext(TabContext)
  return (
    <button
      style={{ fontWeight: activeTab === id ? 'bold' : 'normal' }}
      onClick={() => setActiveTab(id)}
    >
      {children}
    </button>
  )
}

function TabPanel({ id, children }) {
  const { activeTab } = useContext(TabContext)
  if (activeTab !== id) return null
  return <div className="tab-panel">{children}</div>
}

// 네임스페이스로 묶기
Tabs.List = TabList
Tabs.Tab = Tab
Tabs.Panel = TabPanel

// 사용 — 직관적인 API
export default function App() {
  return (
    <Tabs defaultTab="info">
      <Tabs.List>
        <Tabs.Tab id="info">정보</Tabs.Tab>
        <Tabs.Tab id="review">리뷰</Tabs.Tab>
      </Tabs.List>
      <Tabs.Panel id="info"><p>상품 정보</p></Tabs.Panel>
      <Tabs.Panel id="review"><p>리뷰 목록</p></Tabs.Panel>
    </Tabs>
  )
}`,
  }}
/>

## Render Props 패턴

컴포넌트가 렌더링할 내용을 **함수 props**로 받아 유연하게 UI를 결정합니다.

<CodePlayground
  template="react"
  activeFile="/App.js"
  label="Render Props 실습 — 상자 안에서 마우스를 움직여보세요"
  files={{
    '/App.js': `import { useState } from 'react'

// 마우스 위치를 추적하는 로직 분리
function MouseTracker({ render }) {
  const [position, setPosition] = useState({ x: 0, y: 0 })

  function handleMouseMove(e) {
    const rect = e.currentTarget.getBoundingClientRect()
    setPosition({ x: e.clientX - rect.left, y: e.clientY - rect.top })
  }

  return (
    <div onMouseMove={handleMouseMove} style={{ height: '200px', border: '1px solid gray', position: 'relative' }}>
      {render(position)}
    </div>
  )
}

// 사용 — 표시 방식만 달리
export default function App() {
  return (
    <MouseTracker render={({ x, y }) => (
      <>
        <p>좌표: {Math.round(x)}, {Math.round(y)}</p>
        <span style={{ position: 'absolute', left: x, top: y }}>●</span>
      </>
    )} />
  )
}`,
  }}
/>

## 컴포넌트 설계 원칙

**단일 책임 원칙**: 컴포넌트 하나는 한 가지 일만 합니다.

```jsx
// ❌ 한 컴포넌트가 너무 많은 일을 함
function UserDashboard() {
  // 사용자 정보 + 통계 + 알림 + 설정을 모두 처리
}

// ✅ 역할별로 분리
function UserDashboard() {
  return (
    <div>
      <UserProfile />   {/* 사용자 정보 */}
      <Statistics />    {/* 통계 */}
      <Notifications /> {/* 알림 */}
    </div>
  )
}
```

**props interface 최소화**: 필요한 props만 받습니다.

```jsx
// ❌ 유저 객체 전체를 넘기면 컴포넌트가 어떤 정보를 쓰는지 불명확
function UserBadge({ user }) {
  return <span>{user.name}</span>
}

// ✅ 필요한 값만 받기
function UserBadge({ name }) {
  return <span>{name}</span>
}
```

## 핵심 정리

- **합성 우선**: 상속이 아닌 `children`과 props 슬롯으로 컴포넌트를 조합.
- **Compound Component**: 관련 컴포넌트를 그룹으로 묶어 직관적인 API 제공. Context로 내부 상태 공유.
- **Render Props**: 렌더링 로직을 함수 props로 분리. 현재는 커스텀 훅이 더 많이 사용.
- 컴포넌트는 하나의 역할, 필요한 props만 받는 방향으로 설계.
