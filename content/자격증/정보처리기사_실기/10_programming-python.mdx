```python
# 자료형
a = 10        # int

# 컨테이너
lst = [1, 2, 3]        # list (mutable, 순서 있음)
tup = (1, 2, 3)        # tuple (immutable)

d = {"name": "홍길동"} # dict (키-값 쌍)
```

## 리스트 조작

```python
nums = [1, 2, 3, 4, 5]

# 슬라이싱 [start:end:step]
print(nums[1:4])    # [2, 3, 4]
print(nums[:3])     # [1, 2, 3]
print(nums[::2])    # [1, 3, 5]
print(nums[::-1])   # [5, 4, 3, 2, 1]  역순

# 리스트 메서드
nums.append(6)      # [1, 2, 3, 4, 5, 6]
nums.pop()          # 마지막 요소 6 제거
nums.sort()         # 정렬
nums.reverse()      # 역순
len(nums)           # 길이
```

## 딕셔너리

```python
person = {"name": "홍길동", "age": 25, "city": "서울"}

# 접근
print(person["name"])          # 홍길동
print(person.get("age", 0))    # 25

# 수정
person["age"] = 26

# 순회
for key, value in person.items():
    print(f"{key}: {value}")

# 키/값 목록
print(list(person.keys()))     # ['name', 'age', 'city']
print(list(person.values()))   # ['홍길동', 26, '서울']
```

## 함수와 람다

```python
# 일반 함수
def add(a, b):
    return a + b

# 기본값 매개변수
def greet(name, msg="안녕하세요"):
    print(f"{msg}, {name}!")

# 람다 (익명 함수)
square = lambda x: x ** 2
print(square(4))     # 16

# 고차 함수
nums = [1, 2, 3, 4, 5]
evens = list(filter(lambda x: x % 2 == 0, nums))  # [2, 4]
squares = list(map(lambda x: x**2, nums))          # [1, 4, 9, 16, 25]
total = sum(nums)                                   # 15
```

## 리스트 컴프리헨션

```python
# [표현식 for 변수 in iterable if 조건]
squares = [x**2 for x in range(1, 6)]       # [1, 4, 9, 16, 25]
evens = [x for x in range(10) if x % 2 == 0]  # [0, 2, 4, 6, 8]
```

## 클래스

```python
class Person:
    count = 0  # 클래스 변수

    def __init__(self, name, age):
        self.name = name     # 인스턴스 변수
        self.age = age
        Person.count += 1

    def greet(self):
        return f"안녕, {self.name}!"

    @classmethod
    def get_count(cls):
        return cls.count

p = Person("홍길동", 25)
print(p.greet())           # 안녕, 홍길동!
print(Person.get_count())  # 1
```
