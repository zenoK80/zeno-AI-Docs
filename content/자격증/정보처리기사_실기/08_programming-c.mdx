| 자료형 | 크기 | 범위 |
|--------|------|------|
| int | 4 바이트 | -2³¹ ~ 2³¹-1 |
| char | 1 바이트 | 0 ~ 255 (ASCII) |

| double | 8 바이트 | 소수점 15자리 |

## 포인터 (Pointer)

```c
int a = 10;
int *p = &a;   // p는 a의 주소를 저장

printf("%d", a);    // 10 (a의 값)
printf("%p", p);    // a의 주소값
printf("%d", *p);   // 10 (p가 가리키는 값 = a의 값)

*p = 20;            // a의 값을 20으로 변경
printf("%d", a);    // 20
```

## 배열

```c
int arr[5] = {10, 20, 30, 40, 50};

// 인덱스로 접근
printf("%d", arr[0]);   // 10
printf("%d", arr[2]);   // 30

// 배열명은 첫 번째 원소의 주소
int *p = arr;
printf("%d", *p);       // 10
printf("%d", *(p+1));   // 20
```

## 함수와 재귀

```c
// 팩토리얼 재귀 함수
int factorial(int n) {
    if (n == 0 || n == 1) return 1;
    return n * factorial(n - 1);
}
// factorial(5) = 5 * 4 * 3 * 2 * 1 = 120

// 피보나치 재귀
int fib(int n) {
    if (n <= 1) return n;
    return fib(n-1) + fib(n-2);
}
// fib(5) = 5
```

## 구조체 (struct)

```c
struct Student {
    char name[20];
    int age;
    float gpa;
};

struct Student s;
strcpy(s.name, "홍길동");
s.age = 20;
s.gpa = 3.8;

printf("%s, %d", s.name, s.age);  // 홍길동, 20
```

## 코드 추적 예제

```c
#include <stdio.h>

int main() {
    int i, sum = 0;
    for (i = 1; i <= 5; i++) {
        sum += i;
    }
    printf("sum = %d\n", sum);   // sum = 15
    return 0;
}
```

```c
// 포인터 스왑
void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int x = 3, y = 7;
swap(&x, &y);
printf("%d %d", x, y);  // 7 3
```
